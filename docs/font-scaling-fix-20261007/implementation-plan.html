/**
 * 把绝对单位字号/行高换算为精确 rem（16px 基准）。
 *
 * 为什么必须精确：所有 px 取值都是 0.5 的整数倍，÷16 得到有限的十进制小数，
 * 因此在根字号 100% 时换算后的计算值与改动前逐位相同 → 零视觉回归。
 * 若 px 行高改成无单位比值，像 20px/13px 这类比例无法写成精确小数，
 * 会让 100% 下的计算行高变成 19.994px，破坏等效性。故一律用 rem。
 *
 * 只改字号与行高，不动布局尺寸、图标尺寸与颜色。
 * 幂等：重复运行不再产生变化。
 */
import { readFileSync, writeFileSync, readdirSync, statSync } from "node:fs"
import { join, relative } from "node:path"
import { pathToFileURL } from "node:url"

/** 精确换算：0.5 的整数倍 → 最多 5 位小数的有限小数。 */
export function pxToRem(px) {
  const value = Number(px)
  if (!Number.isFinite(value) || value <= 0) throw new Error(`非法 px 值: ${px}`)
  const rem = value / 16
  if (!Number.isFinite(rem)) throw new Error(`换算失败: ${px}`)
  return `${Number(rem.toFixed(6))}rem`
}

const CSS_COMMENT = /\/\*[\s\S]*?\*\//g

/** 只对注释之外的片段做替换，避免改写注释里的示例。 */
function mapOutsideComments(text, fn) {
  let out = ""
  let last = 0
  for (const m of text.matchAll(CSS_COMMENT)) {
    out += fn(text.slice(last, m.index)) + m[0]
    last = m.index + m[0].length
  }
  return out + fn(text.slice(last))
}

function convertCssChunk(chunk) {
  let out = chunk
  // font 简写：仅当字号/行高带 px 时换算；字重、字族、无单位行高原样保留
  out = out.replace(
    /\bfont:\s*([^;{}]*?)([\d.]+)px(\s*\/\s*([\d.]+)px)?/g,
    (_m, head, size, slash, lh) =>
      `font: ${head}${pxToRem(size)}${slash ? `/${pxToRem(lh)}` : ""}`,
  )
  out = out.replace(/font-size:\s*([\d.]+)px/g, (_m, v) => `font-size: ${pxToRem(v)}`)
  out = out.replace(/line-height:\s*([\d.]+)px/g, (_m, v) => `line-height: ${pxToRem(v)}`)
  return out
}

export function convertCss(text) {
  return mapOutsideComments(text, convertCssChunk)
}

export function convertTsx(text) {
  // 只匹配 text-[Npx]；min-w-/max-w-/gap-/颜色任意值不匹配
  return mapOutsideComments(text, (chunk) =>
    chunk.replace(/text-\[([\d.]+)px\]/g, (_m, v) => `text-[${pxToRem(v)}]`),
  )
}

function walk(dir, out = []) {
  for (const entry of readdirSync(dir)) {
    const p = join(dir, entry)
    if (statSync(p).isDirectory()) walk(p, out)
    else if (/\.(css|tsx|ts)$/.test(entry)) out.push(p)
  }
  return out
}

export function convertFile(absPath) {
  const original = readFileSync(absPath, "utf8")
  const isCss = absPath.endsWith(".css")
  const converted = isCss ? convertCss(original) : convertTsx(original)
  return { original, converted, changed: original !== converted }
}

function main() {
  const dryRun = process.argv.includes("--dry-run")
  const check = process.argv.includes("--check")
  const root = join(process.cwd(), "src")
  let files = 0
  let changedFiles = 0

    files++

  }
  console.log(`\n  扫描 ${files} 个文件，${changedFiles} 个需换算`)
  if (check && changedFiles > 0) process.exitCode = 1
}
