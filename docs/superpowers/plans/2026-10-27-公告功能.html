/**
 * 官网公告（announcements.json）的读取与归一化。
 *
 * 为什么所有校验都收在这个文件里：JSON 由人手工维护，字段错、id 重复、
 * 时间写成"2026年10月27日"都是常态。客户端的原则是**坏数据只丢那一条**，
 * 绝不能因为一条写错就让整个公告列表消失。
 */

export interface Announcement {
  id: string
  title: string
  body: string
  /** epoch ms */
  publishedAt: number
  pinned: boolean
  url?: string
}

export interface AnnouncementsResult {
  items: Announcement[]
  fetchedAt: number
}

const MAX_TITLE_LENGTH = 60
const MAX_BODY_LENGTH = 2000
const MAX_ITEMS = 50

function asTrimmedString(value: unknown, limit: number): string {
  if (typeof value !== "string") return ""
  return value.trim().slice(0, limit)
}

/** 结构合法（是数组或带 announcements 数组的对象）才返回数组，否则 null。 */
export function readAnnouncementList(payload: unknown): unknown[] | null {
  if (Array.isArray(payload)) return payload
  if (payload && typeof payload === "object") {
    const list = (payload as { announcements?: unknown }).announcements
    if (Array.isArray(list)) return list
  }
  return null
}

export function normalizeAnnouncements(payload: unknown): Announcement[] {
  const list = readAnnouncementList(payload)
  if (!list) return []

  const seen = new Set<string>()
  const items: Announcement[] = []
  for (const raw of list) {
    if (!raw || typeof raw !== "object") continue
    const record = raw as Record<string, unknown>
    const id = asTrimmedString(record.id, 64)
    const title = asTrimmedString(record.title, MAX_TITLE_LENGTH)
    const body = asTrimmedString(record.body, MAX_BODY_LENGTH)
    const publishedAt = typeof record.publishedAt === "string" ? Date.parse(record.publishedAt) : Number.NaN
    if (!id || !title || !body || Number.isNaN(publishedAt)) continue
    if (seen.has(id)) continue
    seen.add(id)

    const rawUrl = asTrimmedString(record.url, 2048)
    const url = /^https?:\/\//i.test(rawUrl) ? rawUrl : undefined

    items.push({ id, title, body, publishedAt, pinned: record.pinned === true, url })
  }

  items.sort((a, b) => {
    if (a.pinned !== b.pinned) return a.pinned ? -1 : 1
    return b.publishedAt - a.publishedAt
  })
  return items.slice(0, MAX_ITEMS)
}

/** 列表里的日期：同年 "10-27"，跨年 "2025-12-31"。 */
export function formatAnnouncementDate(timestamp: number, now = Date.now()): string {
  const date = new Date(timestamp)
  const pad = (value: number) => String(value).padStart(2, "0")
  const monthDay = `${pad(date.getMonth() + 1)}-${pad(date.getDate())}`
  if (date.getFullYear() === new Date(now).getFullYear()) return monthDay
  return `${date.getFullYear()}-${monthDay}`
}

/** 相对时间：刚刚 / N 分钟前 / N 小时前 / 日期。 */
export function formatRelativeTime(timestamp: number, now = Date.now()): string {
  if (!Number.isFinite(timestamp)) return ""
  const diff = now - timestamp
  if (diff < 0) return "刚刚"
  if (diff < 60_000) return "刚刚"
  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)} 分钟前`
  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)} 小时前`
  return formatAnnouncementDate(timestamp, now)
}

Step 3: 实现

import { getHttpFetch } from "@/lib/tauri-fetch"

export const ANNOUNCEMENTS_URLS = [
  "https://www.qmai.pro/announcements.json",
  "https://qmai.pro/announcements.json",
]

const FETCH_TIMEOUT_MS = 8_000

/**
 * 拉一次公告。**永远不抛**：失败返回 null，调用方保留本地缓存。
 * 备用域名是为了应对 CDN/域名单点故障，不是重试同一个地址。
 */
export async function fetchAnnouncements(options: {
  urls?: string[]
  timeoutMs?: number
} = {}): Promise<AnnouncementsResult | null> {
  const urls = options.urls ?? ANNOUNCEMENTS_URLS
  const timeoutMs = options.timeoutMs ?? FETCH_TIMEOUT_MS
  const httpFetch = await getHttpFetch()

  for (const url of urls) {
    const controller = new AbortController()
    const timer = setTimeout(() => controller.abort(), timeoutMs)
    try {
      const response = await httpFetch(url, {
        headers: { Accept: "application/json" },
        signal: controller.signal,
      })
      if (!response.ok) continue
      const payload: unknown = await response.json()
      const list = readAnnouncementList(payload)
      if (!list) continue
      return { items: normalizeAnnouncements(list), fetchedAt: Date.now() }
    } catch (error) {
      console.warn(`[announcements] 拉取失败：${url}`, error)
    } finally {
      clearTimeout(timer)
    }
  }
  return null
}

/**

 *
 * 三条硬规则：
 * 1. 拉取失败**不清空**已有列表 —— 断网时用户仍应看到上次的公告。
 * 2. 已读只增不减（除按当前列表修剪），收起/重开不会退回未读。
 * 3. 持久化失败只 warn，不影响界面 —— 公告不是关键数据。
 */

import { create } from "zustand"

import { fetchAnnouncements, normalizeAnnouncements, type Announcement } from "@/lib/announcements"
import { getStore } from "@/lib/web-store"

export const ANNOUNCEMENTS_CACHE_KEY = "announcementsCache"
export const ANNOUNCEMENT_READ_IDS_KEY = "announcementReadIds"

const MAX_READ_IDS = 200

interface CachedAnnouncements {
  items: Announcement[]
  fetchedAt: number
}

export interface AnnouncementStoreState {
  items: Announcement[]
  readIds: Set<string>
  lastFetchedAt: number | null
  status: "idle" | "loading" | "ready" | "error"
  dialogOpen: boolean
  hydrate: () => Promise<void>
  refresh: () => Promise<void>
  openDialog: () => void
  closeDialog: () => void
  markRead: (id: string) => Promise<void>
  markAllRead: () => Promise<void>
}

/** 未读数：放在模块级，保证 selector 引用稳定（否则每次 render 都可能重算）。 */
export const selectUnreadCount = (state: AnnouncementStoreState): number =>
  state.items.reduce((count, item) => (state.readIds.has(item.id) ? count : count + 1), 0)

async function persist(items: Announcement[], readIds: Set<string>, fetchedAt: number | null): Promise<void> {
  try {
    const store = await getStore()
    await store.set(ANNOUNCEMENTS_CACHE_KEY, { items, fetchedAt: fetchedAt ?? Date.now() })
    await store.set(ANNOUNCEMENT_READ_IDS_KEY, [...readIds].slice(-MAX_READ_IDS))
  } catch (error) {
    console.warn("[announcements] 保存公告状态失败：", error)
  }
}

function readIdsFrom(value: unknown, items: Announcement[]): Set<string> {
  if (!Array.isArray(value)) return new Set()
  const known = new Set(items.map((item) => item.id))
  return new Set(value.filter((id): id is string => typeof id === "string" && known.has(id)).slice(-MAX_READ_IDS))
}

let inflight: Promise<void> | null = null

export const useAnnouncementStore = create<AnnouncementStoreState>((set, get) => ({
  items: [],
  readIds: new Set(),
  lastFetchedAt: null,
  status: "idle",
  dialogOpen: false,

  hydrate: async () => {
    try {
      const store = await getStore()
      const cache = await store.get<CachedAnnouncements>(ANNOUNCEMENTS_CACHE_KEY)
      const items = normalizeAnnouncements(cache?.items)
      const readIds = readIdsFrom(await store.get<unknown>(ANNOUNCEMENT_READ_IDS_KEY), items)
      const fetchedAt = typeof cache?.fetchedAt === "number" ? cache.fetchedAt : null
      set({ items, readIds, lastFetchedAt: fetchedAt, status: items.length > 0 ? "ready" : "idle" })
    } catch (error) {
      console.warn("[announcements] 读取公告缓存失败：", error)
    }
  },

  refresh: async () => {
    if (inflight) return inflight
    inflight = (async () => {
      set({ status: "loading" })
      const result = await fetchAnnouncements()
      if (!result) {
        set({ status: "error" })
        return
      }
      const readIds = readIdsFrom([...get().readIds], result.items)
      set({ items: result.items, readIds, lastFetchedAt: result.fetchedAt, status: "ready" })
      await persist(result.items, readIds, result.fetchedAt)
    })()
    try {
      await inflight
    } finally {
      inflight = null
    }
  },

  openDialog: () => {
    set({ dialogOpen: true })
    void get().refresh()
  },

  closeDialog: () => set({ dialogOpen: false }),

  markRead: async (id) => {
    const { readIds, items, lastFetchedAt } = get()
    if (readIds.has(id)) return
    const next = new Set(readIds)
    next.add(id)
    set({ readIds: next })
    await persist(items, next, lastFetchedAt)
  },

  markAllRead: async () => {
    const { items, readIds, lastFetchedAt } = get()
    const next = new Set(readIds)
    for (const item of items) next.add(item.id)
    set({ readIds: next })
    await persist(items, next, lastFetchedAt)
  },
}))
