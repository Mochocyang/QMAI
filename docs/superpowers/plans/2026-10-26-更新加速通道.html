软件更新加速通道 · 实施计划

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox-style tracking.

Goal: 用户点「检查更新」时，自动在多条通道（直连 / 加速反代 / jsDelivr）里探活择优，用最快且可用的一条下载安装包；全程无感，云同步永远不经过任何第三方中转。

Architecture: 三个互不依赖的 Rust 模块串成流水线——channels（有哪些通道、URL 怎么拼，纯数据）→ probe（并发测 RTT、排序、10 分钟缓存）→ download（按序尝试、失败换下一名、清单与安装包必须同通道）。验签仍交给 tauri-plugin-updater，安全边界不动。

Tech Stack: Rust（reqwest blocking + rustls）+ Tauri 2 命令 + React 19 前端调用；测试用 std::net::TcpListener 起本地 HTTP 替身（照 src-tauri/src/cloud_sync/providers/webdav.rs 末尾的既有写法）。

    impl Channel {
    /// 把一个 GitHub release 地址映射到本通道的地址。
    ///
    /// 返回 `None` 表示**这个通道不支持这类地址**（例如 jsDelivr 代理不了 release 资产）。
    /// 调用方看到 `None` 应当跳过该通道，而不是把它当作「可用但失败」。
    pub fn map_release_url(&self, github_url: &str) -> Option<String> {
        match self.rewrite {
            Rewrite::Identity => Some(github_url.to_string()),
            Rewrite::Prefix => {
                let base = self.base.as_deref()?.trim();
                if base.is_empty() || !base.starts_with("https://") {
                    // 只接受 https：明文 http 的加速通道会被中间设备随意改写。
                    return None;
                }
                Some(format!("{}{}", ensure_trailing_slash(base), github_url))
            }
            Rewrite::Jsdelivr => None,
        }
    }

    /// jsDelivr 这类 CDN 只能取**仓库文件**（不是 release 资产）。
    pub fn map_repo_file_url(
        &self,
        owner: &str,
        repo: &str,
        branch: &str,
        path: &str,
    ) -> Option<String> {
        match self.rewrite {
            Rewrite::Jsdelivr => Some(format!(
                "https://cdn.jsdelivr.net/gh/{owner}/{repo}@{branch}/{path}"
            )),
            Rewrite::Identity => Some(format!(
                "https://raw.githubusercontent.com/{owner}/{repo}/{branch}/{path}"
            )),
            Rewrite::Prefix => {
                let base = self.base.as_deref()?.trim();
                if base.is_empty() || !base.starts_with("https://") {
                    return None;
                }
                Some(format!(
                    "{}{}",
                    ensure_trailing_slash(base),
                    format!("https://raw.githubusercontent.com/{owner}/{repo}/{branch}/{path}")
                ))
            }
        }
    }
}

fn ensure_trailing_slash(value: &str) -> String {
    if value.ends_with('/') {
        value.to_string()
    } else {
        format!("{value}/")
    }
}

    impl ChannelTable {
    /// 解析远程下发的通道表并做清洗。
    ///
    /// 原则：**坏项丢弃、整体可用**。远程表是网络内容，不能被它搞崩更新流程：
    /// 重复 id、明文 http、缺 base、未知 rewrite 一律剔除；剔除后如果连 direct
    /// 都没了，自动补回内置的 direct，保证永远有一条兜底。
    pub fn parse(json: &str) -> Option<Self> {
        let raw: serde_json::Value = serde_json::from_str(json).ok()?;
        let version = raw.get("version")?.as_u64()? as u32;
        let items = raw.get("channels")?.as_array()?.clone();

        let mut out: Vec<Channel> = Vec::new();
        for item in items {
            let Ok(channel) = serde_json::from_value::<Channel>(item) else {
                continue;
            };
            if channel.id.trim().is_empty() {
                continue;
            }
            if out.iter().any(|existing| existing.id == channel.id) {
                continue;
            }
            if channel.rewrite == Rewrite::Prefix {
                let Some(base) = channel.base.as_deref() else {
                    continue;
                };
                if !base.trim().starts_with("https://") {
                    continue;
                }
            }
            out.push(channel);
        }

        let direct = Self::builtin()
            .channels
            .into_iter()
            .find(|channel| channel.id == "direct")?;
        if !out.iter().any(|channel| channel.id == "direct") {
            out.insert(0, direct);
        }
        Some(Self {
            version,
            channels: out,
        })
    }
}

    //! 并发探活：测出哪条通道能通、多快。
//!
//! 探测对象是 `latest.json` 这种几百字节的小文件 —— 既验证"连通"，
//! 也验证"这个通道真的能取到更新清单"（jsDelivr 对 release 资产会 404）。

use std::time::{Duration, Instant};

/// 连接阶段最多等这么久；超时立刻判死换下一名，不傻等到总超时。
pub const PROBE_CONNECT_TIMEOUT: Duration = Duration::from_secs(3);
pub const PROBE_TOTAL_TIMEOUT: Duration = Duration::from_secs(5);
/// 探活结果缓存 10 分钟。
pub const PROBE_TTL_MS: i64 = 10 * 60 * 1000;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ProbeResult {
    pub channel_id: String,
    pub available: bool,
    pub rtt_ms: Option<u64>,
}

#[derive(Debug, Default)]
pub struct ProbeCache {
    content: Vec<ProbeResult>,
    at_ms: i64,
}

impl ProbeCache {
    pub fn store(&mut self, results: Vec<ProbeResult>, now_ms: i64) {
        self.content = results;
        self.at_ms = now_ms;
    }

    pub fn fresh(&self, now_ms: i64) -> Option<&[ProbeResult]> {
        if self.content.is_empty() || now_ms - self.at_ms > PROBE_TTL_MS {
            return None;
        }
        Some(&self.content)
    }
}

pub fn client() -> Result<reqwest::blocking::Client, String> {
    reqwest::blocking::Client::builder()
        .connect_timeout(PROBE_CONNECT_TIMEOUT)
        .timeout(PROBE_TOTAL_TIMEOUT)
        .user_agent("QMaiWrite-Updater")
        .build()
        .map_err(|error| format!("updater.client:{error}"))
}

/// 探一条 URL。任何失败都只是"不可用"，不是错误。
pub fn probe_url(client: &reqwest::blocking::Client, url: &str) -> ProbeResult {
    let started = Instant::now();
    let available = client
        .get(url)
        .send()
        .map(|response| response.status().is_success())
        .unwrap_or(false);
    ProbeResult {
        channel_id: String::new(),
        available,
        rtt_ms: if available {
            Some(started.elapsed().as_millis() as u64)
        } else {
            None
        },
    }
}

/// 下载尝试顺序：可用的按 RTT 升序，不可用的排后面，`direct` 永远垫底。
///
/// 为什么 direct 垫底：它是唯一不依赖第三方、也永远不会"服务下线"的那条。
/// 即使探活时它很慢甚至失败，也留着最后一次尝试机会。
pub fn order_for_download(mut probed: Vec<ProbeResult>) -> Vec<ProbeResult> {
    probed.sort_by_key(|item| {
        let is_direct = item.channel_id == "direct";
        (
            if item.available && !is_direct { 0 } else if !is_direct { 1 } else { 2 },
            item.rtt_ms.unwrap_or(u64::MAX),
        )
    });
    probed
}

    //! 按通道顺序下载。第一个成功的通道胜出，其余不再尝试。

/// 一个通道对本次下载的"可用形态"。
///
/// `manifest_url` 为 `None` 表示这个通道不支持当前文件类型（jsDelivr 之于 release 资产）。
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ChannelAttempt {
    pub channel_id: String,
    pub label: String,
    pub manifest_url: Option<String>,
}

/// 依序尝试下载，返回 `(数据, 命中的通道 id)`。
///
/// 失败原因全部收集进最终错误里：用户报障时能直接看出"是哪几条通道坏了、为什么"。
pub fn download_with_fallback<F>(
    attempts: Vec<ChannelAttempt>,
    mut fetch: F,
) -> Result<(Vec<u8>, String), String>
where
    F: FnMut(&ChannelAttempt) -> Result<Vec<u8>, String>,
{
    let mut reasons: Vec<String> = Vec::new();
    for attempt in attempts {
        if attempt.manifest_url.is_none() {
            continue;
        }
        match fetch(&attempt) {
            Ok(bytes) => return Ok((bytes, attempt.channel_id)),
            Err(reason) => reasons.push(format!("{}: {reason}", attempt.label)),
        }
    }
    if reasons.is_empty() {
        return Err("没有可用的下载通道".to_string());
    }
    Err(format!("全部通道都失败了（{}）", reasons.join("；")))
}

    /// 当前生效的代理 URL；没启用或地址非法时返回 `None`。
///
/// 为什么需要它：`tauri-plugin-updater` 走自己的 http 客户端，**不读**我们写进
/// 环境变量的 `HTTP_PROXY`。所以更新通道必须显式把代理交给它，否则用户开了代理
/// 也对「检查更新」无效（这是实测发现的一个漏洞）。
pub fn current_proxy_url(config: &ProxyConfig) -> Option<String> {
    let url = config.url.trim();
    if !config.enabled || url.is_empty() {
        return None;
    }
    if !url.starts_with("http://") && !url.starts_with("https://") {
        return None;
    }
    Some(url.to_string())
}

/// 本次更新的官方清单地址（release 上的 latest.json）。
pub const MANIFEST_URL: &str =

/// 把「通道表 + 探活结果」变成「下载尝试顺序」。
///
/// 每条通道自己决定能不能映射这个清单地址：jsDelivr 映射不了 release 资产，
/// 于是它的 `manifest_url` 是 `None`，下载阶段会直接跳过 —— 绝不会拿它去撞 404。
pub fn build_plan(table: &ChannelTable, probed: Vec<probe::ProbeResult>) -> Vec<ChannelAttempt> {
    let ordered = probe::order_for_download(probed);
    let mut plan: Vec<ChannelAttempt> = Vec::new();
    for result in &ordered {
        if let Some(channel) = table.channels.iter().find(|item| item.id == result.channel_id) {
            plan.push(attempt_for(channel));
        }
    }
    // 兜底：探活列表里没有的通道（例如缓存是旧的）也要补进来，direct 永远垫底。
    for channel in &table.channels {
        if plan.iter().any(|item| item.channel_id == channel.id) {
            continue;
        }
        if channel.id == "direct" {
            plan.push(attempt_for(channel));
        }
    }
    plan
}

fn attempt_for(channel: &Channel) -> ChannelAttempt {
    ChannelAttempt {
        channel_id: channel.id.clone(),
        label: channel.label.clone(),
        manifest_url: channel.map_release_url(MANIFEST_URL),
    }
}

    //! 给前端用的更新通道命令：返回「按推荐顺序排好的通道列表」。

use serde::Serialize;
use tauri::{AppHandle, Manager, Runtime};

use crate::updater::{self, channels::ChannelTable, probe};

#[derive(Debug, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct ChannelPlanItem {
    pub channel_id: String,
    pub label: String,
    pub manifest_url: Option<String>,
    pub rtt_ms: Option<u64>,
}

/// 探活并返回下载顺序。前端拿第一项的 `manifestUrl` 去检查更新。
#[tauri::command]
pub async fn updater_channel_plan<R: Runtime>(
    app: AppHandle<R>,
) -> Result<Vec<ChannelPlanItem>, String> {
    let _ = app;
    let table = ChannelTable::builtin();
    let client = probe::client()?;
    let targets: Vec<(String, String)> = table
        .channels
        .iter()
        .filter_map(|channel| {
            channel
                .map_release_url(updater::MANIFEST_URL)
                .map(|url| (channel.id.clone(), url))
        })
        .collect();

    let probed = tauri::async_runtime::spawn_blocking(move || {
        let mut out = Vec::with_capacity(targets.len());
        for (id, url) in targets {
            let mut result = probe::probe_url(&client, &url);
            result.channel_id = id;
            out.push(result);
        }
        out
    })
    .await
    .map_err(|error| format!("探活失败：{error}"))?;

    let plan = updater::build_plan(&table, probed.clone());
    Ok(plan
        .into_iter()
        .map(|item| {
            let rtt = probed
                .iter()
                .find(|result| result.channel_id == item.channel_id)
                .and_then(|result| result.rtt_ms);
            ChannelPlanItem {
                channel_id: item.channel_id,
                label: item.label,
                manifest_url: item.manifest_url,
                rtt_ms: rtt,
            }
        })
        .collect())
}

    import { invoke } from "@tauri-apps/api/core"

import { isTauri } from "@/lib/platform"

export interface UpdateChannelPlanItem {
  channelId: string
  label: string
  /** `null` 表示这条通道不支持 release 资产（例如 jsDelivr）。 */
  manifestUrl: string | null
  rttMs: number | null
}

/**
 * 选出本次检查更新应该用的通道。
 *
 * 返回 `null` 表示没有任何可用通道 —— 调用方要退回 tauri.conf.json 里的官方端点，
 * **不能因为探活失败就让更新彻底不可用**。
 */
export async function resolveUpdateChannel(): Promise {
  if (!isTauri()) return null
  try {
    const plan = await invoke("updater_channel_plan")
    return plan.find((item) => Boolean(item.manifestUrl)) ?? null
  } catch (error) {
    console.warn("[updater] 探活失败，退回官方端点：", error)
    return null
  }
}
