use std::ffi::OsString;
use std::path::{Path, PathBuf};

/// 环境变量覆盖（开发调试用，对应设计文档 8.1）
pub const HOME_ENV: &str = "QMAI_PLUGIN_HOME";

/// 插件根目录：`QMAI_PLUGIN_HOME` 优先，否则 `<app_data>/plugins`。
pub fn plugins_root() -> PathBuf {
    let env = std::env::var_os(HOME_ENV);
    let base = crate::mcp::paths::app_data_dir();
    plugins_root_with(env, base)
}

fn plugins_root_with(env: Option<OsString>, app_data: PathBuf) -> PathBuf {
    if let Some(v) = env {
        // 只借用 lossy 结果判空：非 UTF-8 路径若被有损转换再重建，会指向一个不存在的目录
        if !v.to_string_lossy().trim().is_empty() {
            return PathBuf::from(v);
        }
    }
    app_data.join("plugins")
}

pub fn plugin_dir(root: &Path, id: &str) -> PathBuf {
    root.join(id)
}
// id 允许连字符，所以 `my-data` 这个 id 会与插件 `my` 的 `-data` 目录撞路径。
// 改用 `.` 分隔：id 正则不允许 `.`，`{id}.data` 永远不会等于任何 `{id'}`
pub fn plugin_data_dir(root: &Path, id: &str) -> PathBuf {
    root.join(format!("{id}.data"))
}
pub fn staging_dir(root: &Path) -> PathBuf {
    root.join(".staging")
}
pub fn registry_file(root: &Path) -> PathBuf {
    root.join(".registry.json")
}

/// 清单校验错误（code 对应设计文档 4.2 的错误码列）
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ManifestError { pub code: &'static str, pub detail: String }

impl ManifestError {
    pub fn code(&self) -> &'static str { self.code }
    fn new(code: &'static str, detail: impl Into<String>) -> Self {
        Self { code, detail: detail.into() }
    }
}

/// 宿主暴露给插件的服务名白名单（第一期只开这两个）
pub const KNOWN_INJECT: &[&str] = &["host.qmai.tools", "host.logger"];
/// 允许申请的权限（data:write 自动授予，不必声明）
pub const KNOWN_PERMISSIONS: &[&str] =
    &["fs:project-read", "fs:project-write", "net:https", "llm:use"];

fn is_valid_id(id: &str) -> bool {
    let b = id.as_bytes();
    if b.len() < 2 || b.len() > 64 { return false; }
    let first_ok = b[0].is_ascii_lowercase() || b[0].is_ascii_digit();
    first_ok && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn is_valid_semver(v: &str) -> bool {
    let parts: Vec<&str> = v.split('.').collect();
    parts.len() == 3 && parts.iter().all(|p| !p.is_empty() && p.chars().all(|c| c.is_ascii_digit()))
}

fn is_safe_relative(entry: &str) -> bool {
    if entry.is_empty() { return false; }
    let p = std::path::Path::new(entry);
    if p.is_absolute() { return false; }
    !entry.split(['/', '\\']).any(|seg| seg == "..")
}

fn is_valid_tool_name(n: &str) -> bool {
    let b = n.as_bytes();
    if b.len() < 3 || b.len() > 64 { return false; }
    b[0].is_ascii_lowercase()
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

impl PluginManifest {
    /// 解析 + 校验。字段级错误返回第一个命中的错误码。
    pub fn parse(raw: &str) -> Result<Self, ManifestError> {
        let m: PluginManifest = serde_json::from_str(raw)
            .map_err(|e| ManifestError::new("E_JSON_INVALID", e.to_string()))?;
        m.validate()?;
        Ok(m)
    }

    fn validate(&self) -> Result<(), ManifestError> {
        if self.schema != 1 {
            return Err(ManifestError::new("E_SCHEMA_UNSUPPORTED", self.schema.to_string()));
        }
        if !is_valid_id(&self.id) {
            return Err(ManifestError::new("E_ID_INVALID", self.id.clone()));
        }
        if self.name.trim().is_empty() || self.name.chars().count() > 64 {
            return Err(ManifestError::new("E_NAME_INVALID", self.name.clone()));
        }
        if !is_valid_semver(&self.version) {
            return Err(ManifestError::new("E_VERSION_INVALID", self.version.clone()));
        }
        if !is_safe_relative(&self.runtime.entry) {
            return Err(ManifestError::new("E_ENTRY_INVALID", self.runtime.entry.clone()));
        }
        if !(1_000..=60_000).contains(&self.runtime.start_timeout_ms) {
            return Err(ManifestError::new(
                "E_START_TIMEOUT_INVALID",
                self.runtime.start_timeout_ms.to_string(),
            ));
        }
        for svc in &self.inject {
            if !KNOWN_INJECT.contains(&svc.as_str()) {
                return Err(ManifestError::new("E_INJECT_UNKNOWN", svc.clone()));
            }
        }
        for p in &self.permissions {
            if !KNOWN_PERMISSIONS.contains(&p.as_str()) {
                return Err(ManifestError::new("E_PERMISSION_UNKNOWN", p.clone()));
            }
        }
        for t in &self.contributes.tools {
            if !is_valid_tool_name(&t.name) {
                return Err(ManifestError::new("E_TOOL_NAME_INVALID", t.name.clone()));
            }
        }
        Ok(())
    }
}

/// 插件注册表：`.registry.json` 是唯一权威（对应 DSH 的单一开关点教训）
pub struct PluginRegistry { file: RegistryFile }

impl PluginRegistry {
    /// 文件不存在 → 空表；文件损坏 → 报错（绝不当作空表覆盖）
    pub fn load(root: &Path) -> Result<Self, String> {
        let path = paths::registry_file(root);
        match std::fs::read_to_string(&path) {
            Ok(text) => {
                let file: RegistryFile = serde_json::from_str(&text)
                    .map_err(|e| format!("插件注册表损坏（{}）：{e}", path.display()))?;
                Ok(Self { file })
            }
            Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
                Ok(Self { file: RegistryFile { version: 1, plugins: BTreeMap::new() } })
            }
            Err(e) => Err(format!("读取插件注册表失败：{e}")),
        }
    }

    pub fn entries(&self) -> impl Iterator<Item = &RegistryEntry> {
        self.file.plugins.values()
    }

    pub fn get(&self, id: &str) -> Option<&RegistryEntry> { self.file.plugins.get(id) }

    pub fn list(&self) -> Vec<RegistryEntry> {
        self.file.plugins.values().cloned().collect()
    }

    pub fn upsert(&mut self, entry: RegistryEntry) {
        self.file.plugins.insert(entry.id.clone(), entry);
    }

    pub fn remove(&mut self, id: &str) -> Option<RegistryEntry> {
        self.file.plugins.remove(id)
    }

    pub fn set_enabled(&mut self, id: &str, enabled: bool) -> Result<(), String> {
        match self.file.plugins.get_mut(id) {
            Some(e) => { e.enabled = enabled; Ok(()) }
            None => Err(format!("插件 {id} 未安装")),
        }
    }

    pub fn save(&self, root: &Path) -> Result<(), String> {
        std::fs::create_dir_all(root).map_err(|e| e.to_string())?;
        let path = paths::registry_file(root);
        let text = serde_json::to_string_pretty(&self.file).map_err(|e| e.to_string())?;
        let tmp = path.with_extension("json.tmp");
        std::fs::write(&tmp, text).map_err(|e| e.to_string())?;
        crate::atomic_file::replace_file_atomically(&tmp, &path)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ProcessState {
    Stopped, Starting, Running, Failed, Crashed, DisabledByCrash,
}

impl ProcessState {
    pub fn can_transition_to(self, next: ProcessState) -> bool {
        use ProcessState::*;
        matches!(
            (self, next),
            (Stopped, Starting) | (Failed, Starting) | (Crashed, Starting)
                | (DisabledByCrash, Starting) | (Starting, Running) | (Starting, Failed)
                | (Running, Crashed) | (Running, Stopped) | (Failed, Stopped)
                | (Crashed, Stopped) | (Running, DisabledByCrash)
        )
    }
    pub fn as_str(self) -> &'static str {
        match self {
            ProcessState::Stopped => "stopped",
            ProcessState::Starting => "starting",
            ProcessState::Running => "running",
            ProcessState::Failed => "failed",
            ProcessState::Crashed => "crashed",
            ProcessState::DisabledByCrash => "disabled_by_crash",
        }
    }
}

/// 崩溃重启预算：1s / 3s / 9s，用尽后停用（设计文档 7.2）
#[derive(Debug, Default)]
pub struct CrashBudget { attempt: u32 }

impl CrashBudget {
    pub fn next_backoff(&mut self) -> Option<Duration> {
        let secs = match self.attempt {
            0 => 1,
            1 => 3,
            2 => 9,
            _ => return None,
        };
        self.attempt += 1;
        Some(Duration::from_secs(secs))
    }
    pub fn reset(&mut self) { self.attempt = 0; }
    pub fn exhausted(&self) -> bool { self.attempt >= 3 }
}

/// 只保留尾部 N 字节的缓冲（防止插件疯狂打印打爆内存）
pub struct TailBuffer { limit: usize, data: VecDeque<u8> }

impl TailBuffer {
    pub fn new(limit: usize) -> Self { Self { limit, data: VecDeque::new() } }
    pub fn push(&mut self, chunk: &[u8]) {
        for b in chunk {
            if self.data.len() == self.limit { self.data.pop_front(); }
            self.data.push_back(*b);
        }
    }
    pub fn as_lossy(&self) -> String {
        String::from_utf8_lossy(&self.data.iter().copied().collect::<Vec<u8>>()).into_owned()
    }
}

pub async fn build_command(
    plugin_dir: &std::path::Path,
    manifest: &super::manifest::PluginManifest,
) -> Result<Command, String> {
    use super::manifest::RuntimeType;
    let entry_path = plugin_dir.join(&manifest.runtime.entry);
    let mut cmd = match manifest.runtime.entry_type {
        RuntimeType::Node => {
            let node = crate::commands::cli_resolver::find_cli_command("node", &["node.exe", "node"])
                .await
                .map_err(|e| format!("找不到 Node（插件 {} 需要）：{e}", manifest.id))?;
            let mut c = Command::new(node);
            c.arg(&entry_path);
            c
        }
        RuntimeType::Python => {
            let py = crate::commands::cli_resolver::find_cli_command("python3", &["python.exe", "python3.exe"])
                .await
                .map_err(|e| format!("找不到 Python（插件 {} 需要）：{e}", manifest.id))?;
            let mut c = Command::new(py);
            c.arg(&entry_path);
            c
        }
        RuntimeType::Binary => Command::new(&entry_path),
    };
    cmd.args(&manifest.runtime.args);
    cmd.current_dir(plugin_dir);
    cmd.stdin(std::process::Stdio::piped());
    cmd.stdout(std::process::Stdio::piped());
    cmd.stderr(std::process::Stdio::piped());
    // Windows：禁止弹出控制台窗口（与 claude_cli 等既有实现一致）
    #[cfg(windows)]
    {
        const CREATE_NO_WINDOW: u32 = 0x0800_0000;

    }
    Ok(cmd)
}

use serde_json::{json, Value};

/// 一条 JSON-RPC 消息（请求或响应，取决于有没有 result/error）
#[derive(Debug, Clone, serde::Deserialize)]
pub struct RpcMessage {
    #[serde(default)]
    pub id: Option<Value>,
    #[serde(default)]
    pub result: Option<Value>,
    #[serde(default)]
    pub error: Option<RpcErrorBody>,
}

#[derive(Debug, Clone, serde::Deserialize)]
pub struct RpcErrorBody { pub code: i64, pub message: String }

#[derive(Debug, Clone)]
pub struct PluginToolDecl {
    pub name: String,
    pub description: String,
    pub input_schema: Value,
}

impl RpcMessage {
    pub fn into_tool_list(self) -> Result<Vec<PluginToolDecl>, String> {
        if let Some(e) = self.error { return Err(format!("{}: {}", e.code, e.message)); }
        let tools = self.result
            .and_then(|r| r.get("tools").cloned())
            .and_then(|t| t.as_array().cloned())
            .ok_or_else(|| "tools/list 响应缺少 result.tools".to_string())?;
        Ok(tools.iter().filter_map(|t| {
            Some(PluginToolDecl {
                name: t.get("name")?.as_str()?.to_string(),
                description: t.get("description").and_then(|d| d.as_str()).unwrap_or_default().to_string(),
                input_schema: t.get("inputSchema").cloned().unwrap_or(json!({"type":"object","properties":{}})),
            })
        }).collect())
    }

    /// tool 调用结果：拼接所有 type=text 的 content
    pub fn into_tool_call(self) -> Result<String, String> {
        if let Some(e) = self.error { return Err(format!("{}: {}", e.code, e.message)); }
        let content = self.result
            .and_then(|r| r.get("content").cloned())
            .and_then(|c| c.as_array().cloned())
            .ok_or_else(|| "tools/call 响应缺少 result.content".to_string())?;
        let texts: Vec<String> = content.iter()
            .filter(|c| c.get("type").and_then(|t| t.as_str()) == Some("text"))
            .filter_map(|c| c.get("text").and_then(|t| t.as_str()).map(str::to_string))
            .collect();
        Ok(texts.join("\n"))
    }
}

pub fn initialize_request(id: u64, plugin_id: &str, host_url: &str, token: &str, granted: &[String]) -> Value {

use std::path::{Component, Path, PathBuf};

/// 检查是否拥有某个权限；`data:write` 永远放行（设计文档 6.2）
pub fn require(granted: &[String], permission: &str) -> Result<(), String> {
    if permission == "data:write" { return Ok(()); }
    if granted.iter().any(|g| g == permission) { Ok(()) }
    else { Err(permission.to_string()) }
}

/// 路径是否落在 root 之内（先做词法归一化，再做前缀比较）
pub fn is_inside(root: &Path, candidate: &str) -> bool {
    let candidate = Path::new(candidate);
    let norm = normalize_lexically(candidate);
    let root_norm = normalize_lexically(root);
    norm.starts_with(&root_norm) && norm != root_norm
}

fn normalize_lexically(p: &Path) -> PathBuf {
    let mut out = PathBuf::new();
    for c in p.components() {
        match c {
            Component::ParentDir => { out.pop(); }
            Component::CurDir => {}
            other => out.push(other.as_os_str()),
        }
    }
    out
}

/// 只允许 https，且拒绝回环与私有网段（防 SSRF）
pub fn is_safe_url(raw: &str) -> bool {
    let Ok(u) = url::Url::parse(raw) else { return false; };
    if u.scheme() != "https" { return false; }
    let Some(host) = u.host_str() else { return false; };
    let Ok(ip) = host.parse::<IpAddr>() else {
        // 域名：允许（不做 DNS 解析，避免额外网络依赖）
        return !host.eq_ignore_ascii_case("localhost");
    };
    !(ip.is_loopback() || ip.is_private() || ip.is_link_local() || ip.is_unspecified())
}

use std::path::{Component, Path, PathBuf};

#[derive(Debug, Clone, PartialEq)]
pub struct SourceSpec { pub owner: String, pub repo: String, pub git_ref: String }

/// 解析用户粘贴的 GitHub 地址（设计文档 8.2 步骤 1）
pub fn parse_source(raw: &str) -> Result<SourceSpec, String> {
    let u = url::Url::parse(raw.trim()).map_err(|e| format!("地址格式不正确：{e}"))?;
    if u.scheme() != "https" { return Err("只支持 https 地址".into()); }
    if u.host_str() != Some("github.com") { return Err("只支持 github.com 的仓库".into()); }
    let path = u.path().trim_matches('/');
    let segs: Vec<&str> = path.split('/').filter(|s| !s.is_empty()).collect();
    if segs.len() != 2 {
        return Err("地址应为 https://github.com/<owner>/<repo>".into());
    }
    let (owner, repo_raw) = (segs[0], segs[1]);
    // 允许 `repo@ref` 形式；也允许 `.git` 后缀
    let (repo, git_ref) = match repo_raw.split_once('@') {
        Some((r, g)) => (r.to_string(), g.to_string()),
        None => (repo_raw.to_string(), "main".to_string()),
    };
    let repo = repo.strip_suffix(".git").unwrap_or(&repo).to_string();
    for seg in [owner, repo.as_str(), git_ref.as_str()] {
        if seg.is_empty() || !seg.chars().all(|c| c.is_ascii_alphanumeric() || "-_.~".contains(c)) {
            return Err(format!("非法字符：{seg}"));
        }
    }
    Ok(SourceSpec { owner: owner.to_string(), repo, git_ref })
}

/// 下载候选：加速通道优先，codeload 直连兜底（与 update-channels.json 同策略）
pub fn download_urls(owner: &str, repo: &str, git_ref: &str) -> Vec<String> {
    let tail = format!("{owner}/{repo}/archive/refs/heads/{git_ref}.zip");
    vec![
        format!("https://gh-proxy.com/https://github.com/{tail}"),
        format!("https://ghproxy.net/https://github.com/{tail}"),
        format!("https://codeload.github.com/{owner}/{repo}/zip/refs/heads/{git_ref}"),
    ]
}

/// zip 条目路径安全校验；不安全返回 None
pub fn is_safe_entry(name: &str) -> Option<PathBuf> {
    let p = Path::new(name);
    if p.is_absolute() { return None; }
    if name.split(['/', '\\']).any(|s| s == "..") { return None; }
    let mut out = PathBuf::new();
    for c in p.components() {
        match c {
            Component::Normal(seg) => out.push(seg),
            Component::CurDir => {}
            _ => return None,
        }
    }
    if out.as_os_str().is_empty() { None } else { Some(out) }
}

use sha2::{Digest, Sha256};

const MAX_ZIP_BYTES: usize = 8 * 1024 * 1024;
const MAX_UNPACKED_BYTES: u64 = 64 * 1024 * 1024;

pub struct InstallOutcome {
    pub plugin_id: String,
    pub version: String,
    pub commit: String,
    pub zip_sha256: String,
}

/// 下载 zip 字节；逐个候选试，全部失败才报错
pub async fn download(client: &reqwest::Client, urls: &[String]) -> Result<Vec<u8>, String> {
    let mut last = String::new();
    for url in urls {
        match client.get(url).send().await {
            Ok(resp) if resp.status().is_success() => {
                let bytes = resp.bytes().await.map_err(|e| format!("读取响应失败：{e}"))?;
                if bytes.len() > MAX_ZIP_BYTES {
                    return Err(format!("插件包超过 {} MB 上限", MAX_ZIP_BYTES / 1024 / 1024));
                }
                return Ok(bytes.to_vec());
            }
            Ok(resp) => last = format!("{url} → HTTP {}", resp.status()),
            Err(e) => last = format!("{url} → {e}"),
        }
    }
    Err(format!("所有下载通道均失败，最后一次：{last}"))
}

/// 解压到 dest：逐条目校验，累加大小，拒绝穿越与符号链接
pub fn extract_zip(bytes: &[u8], dest: &Path) -> Result<(), String> {
    use std::io::Read;
    let mut zip = zip::ZipArchive::new(std::io::Cursor::new(bytes))
        .map_err(|e| format!("插件包不是有效 zip：{e}"))?;
    let mut total: u64 = 0;
    for i in 0..zip.len() {
        let mut file = zip.by_index(i).map_err(|e| e.to_string())?;
        if file.is_dir() { continue; }
        let Some(rel) = is_safe_entry(file.name()) else {
            return Err(format!("插件包包含不安全路径：{}", file.name()));
        };
        total += file.size();
        if total > MAX_UNPACKED_BYTES {
            return Err(format!("解压后超过 {} MB 上限", MAX_UNPACKED_BYTES / 1024 / 1024));
        }
        let out_path = dest.join(&rel);
        if let Some(parent) = out_path.parent() { std::fs::create_dir_all(parent).map_err(|e| e.to_string())?; }
        let mut buf = Vec::new();
        file.read_to_end(&mut buf).map_err(|e| e.to_string())?;
        std::fs::write(&out_path, buf).map_err(|e| e.to_string())?;
    }
    Ok(())
}

/// 完整安装：下载 → 解压到 staging → 读清单 → 原子发布
pub async fn install(
    root: &Path,
    raw_source: &str,
) -> Result<InstallOutcome, String> {
    let spec = parse_source(raw_source)?;
    let client = reqwest::Client::builder()
        .user_agent("qmai-plugin-installer")
        .build().map_err(|e| e.to_string())?;
    let urls = download_urls(&spec.owner, &spec.repo, &spec.git_ref);
    let bytes = download(&client, &urls).await?;

    let hash = { let mut h = Sha256::new(); h.update(&bytes); format!("{:x}", h.finalize()) };

    std::fs::create_dir_all(super::paths::staging_dir(root)).map_err(|e| e.to_string())?;
    let staging = super::paths::staging_dir(root).join(uuid::Uuid::new_v4().to_string());
    std::fs::create_dir_all(&staging).map_err(|e| e.to_string())?;

    let result = (|| -> Result<InstallOutcome, String> {
        extract_zip(&bytes, &staging)?;
        // GitHub zip 内有一层 `repo-ref/` 顶层目录，剥掉
        let payload = single_top_level_dir(&staging)
            .ok_or_else(|| "插件包内应只有一个顶层目录".to_string())?;
        let manifest_path = payload.join("plugin.json");
        let raw = std::fs::read_to_string(&manifest_path)
            .map_err(|_| "插件包根目录缺少 plugin.json".to_string())?;
        let manifest = super::manifest::PluginManifest::parse(&raw)?;

        // 目标目录已存在 → 拒绝（更新走任务之外的流程）
        let target = super::paths::plugin_dir(root, &manifest.id);
        if target.exists() { return Err(format!("插件 {} 已安装", manifest.id)); }

        std::fs::rename(&payload, &target).map_err(|e| format!("发布插件目录失败：{e}"))?;
        Ok(InstallOutcome {
            plugin_id: manifest.id.clone(),
            version: manifest.version.clone(),
            commit: spec.git_ref.clone(),
            zip_sha256: hash,
        })
    })();

    let _ = std::fs::remove_dir_all(&staging);
    result
}

fn single_top_level_dir(dir: &Path) -> Option<PathBuf> {
    let mut dirs = std::fs::read_dir(dir).ok()?
        .filter_map(|e| e.ok())
        .filter(|e| e.file_type().map(|t| t.is_dir()).unwrap_or(false));
    let first = dirs.next()?.path();
    if dirs.next().is_some() { return None; }
    Some(first)
}

  /**
   * 按命名空间前缀批量注销外部工具，返回删除数量。
   *
   * 外部工具统一带 `<命名空间>__` 前缀，停用/卸载时只清自己那一批，
   * 不碰内置工具——所以不能用 clear()（那是全清）。
   */
  unregisterPrefix(prefix: string): number {
    // 空前缀会让 startsWith("") 恒为真，等价于 clear()，必须挡住
    if (!prefix) return 0
    let removed = 0
    for (const name of [...this.tools.keys()]) {
      if (name.startsWith(prefix)) {
        this.tools.delete(name)
        removed += 1
      }
    }
    return removed
  }

 * 否则一次插件安装就能替换掉内置能力（ToolRegistry.register 是 Map.set，
 * 覆盖是静默的）。
 */
export function registerPluginTools(
  registry: ToolRegistry,
  descriptors: PluginToolDescriptor[],
  invoke: PluginToolInvoker,
): void {
  for (const d of descriptors) {
    const hostName = toHostToolName(d.pluginId, d.name)
    if (registry.has(hostName)) continue
    const tool: Tool = {
      name: hostName,
      description: d.description,
      category: "action",
      permission: "confirm",
      parameters: d.parameters,
      execute: async (params) => invoke(d.pluginId, d.name, params),
    }
    registry.register(tool)
  }
}

  return registry.unregisterPrefix(pluginToolPrefix(pluginId))
}

fn summary(e: &RegistryEntry) -> PluginSummary {

        version: e.version.clone(),

        granted_permissions: e.granted_permissions.clone(),

pub fn plugins_list(state: State<'_, PluginsState>) -> Result<Vec<PluginSummary>, String> {
    let root = state.root();
    let reg = PluginRegistry::load(&root)?;
    let mut out: Vec<PluginSummary> = reg.entries().map(summary).collect();
    out.sort_by(|a, b| a.id.cmp(&b.id));

#[tauri::command]
pub async fn plugins_install(state: State<'_, PluginsState>, source: String) -> Result<PluginSummary, String> {
    let root = state.root();
    let outcome = installer::install(&root, &source).await?;

    let mut reg = PluginRegistry::load(&root)?;
    let entry = RegistryEntry {
        id: outcome.plugin_id.clone(),
        version: outcome.version.clone(),
        enabled: false,                       // 装完默认关闭，用户勾权限后才启用
        granted_permissions: vec![],
        source: SourceRecord {
            repo: source,
            git_ref: outcome.commit.clone(),
            commit: outcome.commit,
            zip_sha256: outcome.zip_sha256,
            installed_at: chrono::Utc::now().to_rfc3339(),
        },
    };
    let s = summary(&entry);
    reg.upsert(entry);
    reg.save(&root)?;
    Ok(s)
}

#[tauri::command]
pub fn plugins_enable(state: State<'_, PluginsState>, id: String, granted: Vec<String>) -> Result<(), String> {
    let root = state.root();
    let mut reg = PluginRegistry::load(&root)?;
    let entry = reg.get(&id).cloned().ok_or_else(|| format!("插件 {id} 未安装"))?;
    // 权限必须来自清单允许的范围
    for p in &granted {
        if !manifest::KNOWN_PERMISSIONS.contains(&p.as_str()) {
            return Err(format!("未知权限：{p}"));
        }
    }
    let _ = entry;
    reg.set_enabled(&id, true)?;
    if let Some(e) = reg.list().into_iter().find(|e| e.id == id) {
        let mut e2 = e;
        e2.granted_permissions = granted;
        reg.upsert(e2);
    }
    reg.save(&root)
}

#[tauri::command]
pub fn plugins_disable(state: State<'_, PluginsState>, id: String) -> Result<(), String> {
    let root = state.root();
    let mut reg = PluginRegistry::load(&root)?;
    reg.set_enabled(&id, false)?;
    reg.save(&root)
}

#[tauri::command]
pub fn plugins_uninstall(state: State<'_, PluginsState>, id: String, delete_data: bool) -> Result<(), String> {
    let root = state.root();
    let mut reg = PluginRegistry::load(&root)?;
    reg.remove(&id);
    reg.save(&root)?;
    let dir = paths::plugin_dir(&root, &id);
    if dir.exists() {
        std::fs::remove_dir_all(&dir).map_err(|e| format!("删除插件目录失败：{e}"))?;
    }
    if delete_data {
        let data = paths::plugin_data_dir(&root, &id);
        if data.exists() { let _ = std::fs::remove_dir_all(&data); }
    }
    Ok(())
}

    pub async fn stop(mut self) {

            Ok(_) => {}
            Err(_) => { let _ = self.child.kill().await; }
        }
    }

async fn start_plugin(
    state: &PluginsState,
    root: &std::path::Path,
    entry: &RegistryEntry,
) -> Result<(), String> {
    // 已在运行则不重复启动
    if state.running.lock().await.contains_key(&entry.id) {
        return Ok(());
    }
    let dir = paths::plugin_dir(root, &entry.id);
    let raw = std::fs::read_to_string(dir.join("plugin.json"))

    let port = {
        let mut guard = state.host_port.lock().map_err(|_| "状态不可用".to_string())?;
        if guard.is_none() {
            let api_state = host_api::HostApiState {
                sessions: Arc::new(Mutex::new(HashMap::new())),
                log_sink: Arc::new(Mutex::new(Vec::new())),
            };
            *guard = Some(host_api::start(api_state)?);
        }
        guard.unwrap()
    };
    let host_url = format!("http://127.0.0.1:{port}");

    // 每个插件一个 token，绑定身份与权限
    let token = uuid::Uuid::new_v4().to_string();

    // 注册会话到 HostApi
    // 注意：HostApiState 需在首次创建时保存下来；这里为简化用全局单例
    host_api::register_session(host_api::PluginSession {
        plugin_id: entry.id.clone(),
        granted: entry.granted_permissions.clone(),
        project_root: None,           // 第二期接当前打开项目
        data_root: paths::plugin_data_dir(root, &entry.id),
    }, &token)?;

    let mut cmd = supervisor::build_command(&dir, &manifest).await?;
    for (k, v) in supervisor::host_env(&entry.id, &host_url, &token) {
        cmd.env(k, v);
    }
    let child = cmd.spawn().map_err(|e| format!("启动插件进程失败：{e}"))?;

    state.running.lock().await.insert(
        entry.id.clone(),
        supervisor::LaunchHandle {
            plugin_id: entry.id.clone(),
            token,
            counter: supervisor::RequestCounter::default(),
            child,
            state: supervisor::ProcessState::Starting,
        },
    );
    Ok(())
}

    // 启用即启动；启动失败不回滚启用状态，让用户在界面上看到 failed
    if let Err(e) = start_plugin(&state, &root, &entry).await {

    if let Some(handle) = state.running.lock().await.remove(&id) {
