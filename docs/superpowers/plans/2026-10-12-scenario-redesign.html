    /**
 * 「舞台情景」角色头像与配色
 *
 * 纯函数，无任何副作用：同一个 speakerId 在任何时候都得到同样的首字与
 * 同一个色板下标。所以同一角色在同一场、以及日后回顾历史情景时，
 * 颜色都不会变。
 *
 * ⚠️ 色板里的类名必须写成字面量。Tailwind v4 靠内容扫描收集类名，
 * 动态拼接出来的类名不会进入产物 CSS，界面上就会掉色。
 * 这条约束由 scenario-avatar.spec.ts 里的源码级用例守着。
 *
 * 已知取舍：8 个色槽对 10 个角色，撞色不可避免（散列本身也允许撞）。
 * 优先保证的是「同一角色跨场次不变色」，而不是「同一场里两两不同色」——
 * 头像上的首字与名字标签足以区分撞色的两个人。
 */

/** 名字首字（用于圆形头像）。 */
export function scenarioAvatarInitial(name: string): string {
  const trimmed = name.trim()
  if (!trimmed) return "?"
  // 用码点数组而不是 charAt：代理对（如 emoji）不会被截成半个字符。
  // 已知局限：国旗（如 🇨🇳）与 ZWJ 家庭 emoji 按码点切会取到不完整的一个码点。
  // 角色名目前都是中文，等真出现 emoji 名字再处理（要 ES2023 的 Intl.Segmenter）。
  return Array.from(trimmed)[0]
}

/**
 * 主持人的固定配色。
 *
 * 刻意不参与哈希：主持人在这场里是唯一的，给它一个稳定的主色，
 * 比让它随名字换色更能让人一眼认出「这句是串场」。
 */
export const SCENARIO_MODERATOR_TONE: ScenarioSpeakerTone = {
  avatarClass: "bg-primary/15 text-primary",
  nameClass: "text-primary",
}

/**
 * 稳定哈希：同一 seed 永远落在同一个下标。
 *
 * seed 必须是已规范化的角色 id（`ScenarioParticipant.id` /
 * `ScenarioMessage.speakerId`），**不要传显示名** —— 引擎里两者今天同为
 * 角色名，但一旦 id 与名字分家，两处调用点传的东西不一致就会让同一角色
 * 在同一场里显示两种颜色。
 *
 * 这里与 scenarioAvatarInitial 一样先 trim：让带空白的输入得到同样的结果，
 * 避免「首字正确、颜色却取决于有没有空格」这种不一致。
 */
export function scenarioSpeakerToneIndex(seed: string): number {
  const text = seed.trim()
  let hash = 0
  for (const ch of text) {
    // *31 + 码点：hash 始终小于 2^37，Number 精度足够，>>>0 截成 32 位无符号
    hash = (hash * 31 + (ch.codePointAt(0) ?? 0)) >>> 0
  }
  return hash % SCENARIO_SPEAKER_TONES.length
}

/**
 * 取一个角色的配色；isModerator 为真时返回主持人固定样式。
 *
 * seed 的取值要求见 scenarioSpeakerToneIndex。
 */
export function scenarioSpeakerTone(
  seed: string,
  isModerator = false,
): ScenarioSpeakerTone {
  if (isModerator) return SCENARIO_MODERATOR_TONE
  return SCENARIO_SPEAKER_TONES[scenarioSpeakerToneIndex(seed)]
}

 * 「舞台情景」绑定灵魂
 *
 * ── 为什么需要这个模块 ──
 * `bindCharacterAura` 写盘前会查 `hasCharacterProfile`，要求该角色已经在
 * `wiki/entities`（type=entity 且 tags 含 character）或 `wiki/outlines`
 * 里有「人物小传 / 人物设定」，否则直接抛错。
 *
 * 而舞台情景的候选角色是 `extractStoryContent` 从**章节正文**提取的，
 * 提取过程只读不写 —— 所以这些角色大概率还没有人物页，一点「绑定灵魂」
 * 就会被拒。本模块把「就地补一份最小小传再绑」这件事做完。
 *
 * ── 为什么是「先试后补」而不是「先查后补」──
 * `hasCharacterProfile` 除了查实体页/大纲，还会兜底走一次 `searchWiki`
 * （character-aura.ts:737-745）。若在这里用 `listBindableNovelCharacters`
 * 自己预判，会漏掉 searchWiki 这条命中路径，对本来能绑成功的角色弹出
 * 多余的补小传步骤。所以顺序必须是：直接绑 → 失败且原因正是缺小传 → 才补。
 */

import { createDirectory, fileExists, writeFileAtomic } from "@/commands/fs"
import {
  CHARACTER_AURA_BINDING_BLOCK_MESSAGE,
  bindCharacterAura,
} from "@/lib/novel/character-aura"
import { normalizePath } from "@/lib/path-utils"
import type { ScenarioCandidate } from "./scenario-types"

/** 认知范围最多带入条目数，避免一份草稿长得没法读。 */
const DRAFT_COGNITION_LIMIT = 10

/** 性格 / 风格缺省时的占位文案。 */
const DRAFT_PLACEHOLDER = "（待补充）"

/** 文件名安全化。规则与 graph-node-page.ts 的 fileNameFromLabel 一致。 */
export function scenarioProfileFileName(name: string): string {
  return name.replace(/[\\/:*?"<>|]/g, "-").trim() || "未命名角色"
}

/** 人物页的落盘路径。 */
export function scenarioProfilePagePath(projectPath: string, name: string): string {
  const pp = normalizePath(projectPath)
  return `${pp}/wiki/entities/${scenarioProfileFileName(name)}.md`
}

/**
 * 由候选角色**已有**的信息拼出小传草稿。
 *
 * 不调模型、不消耗 token：ScenarioCandidate 里已经有 profile / personality /
 * speakingStyle / knowledgeScope / doesNotKnow，够拼出一份能过校验、也有
 * 参考价值的草稿。用户可以在弹窗里改。
 */
export function buildProfileDraftFromCandidate(candidate: ScenarioCandidate): string {
  const lines: string[] = []

  lines.push(`# ${candidate.name}`)
  lines.push("")
  lines.push("## 基本信息")
  lines.push(`- 姓名：${candidate.name}`)
  if (candidate.auraName) lines.push(`- 灵魂：${candidate.auraName}`)

  lines.push("")
  lines.push("## 性格")
  lines.push(
    candidate.personality.length > 0
      ? candidate.personality.join("、")
      : DRAFT_PLACEHOLDER,
  )

  lines.push("")
  lines.push("## 说话风格")
  lines.push(candidate.speakingStyle.trim() || DRAFT_PLACEHOLDER)

  const knows = candidate.knowledgeScope.slice(0, DRAFT_COGNITION_LIMIT)
  const notKnows = candidate.doesNotKnow.slice(0, DRAFT_COGNITION_LIMIT)
  if (knows.length > 0 || notKnows.length > 0) {
    lines.push("")
    lines.push("## 认知范围")
    if (knows.length > 0) {
      lines.push("### 知道")
      for (const item of knows) lines.push(`- ${item}`)
    }
    if (notKnows.length > 0) {
      lines.push("### 不知道")
      for (const item of notKnows) lines.push(`- ${item}`)
    }
  }

  lines.push("")
  lines.push("<!-- 由「舞台情景」根据已写章节自动整理，可自由修改 -->")
  return lines.join("\n")
}

/**
 * title 的 YAML 标量形式。**订正**：初稿这里写的是 `title: "${name}"`
 * 直接内插原始名，那是个 Critical（见上方 callout）—— 名字含 `"` 或 `\`
 * 会让整段 frontmatter 解析失败，人物页写了却永远绑不上。
 * 含 `"` 或 `\` 时改用单引号标量（其中的 `"`、`\` 是普通字面量，`'` 按
 * YAML 规则加倍）；其余名字保持双引号，这样含 `'` 的名字能被原样取回。
 * 因为 character-aura.ts 的 extractPrimaryTitle 只剥首尾引号、不反转义。
 */
function titleScalar(name: string): string {
  if (/["\\]/.test(name)) return `'${name.replace(/'/g, "''")}'`
  return `"${name}"`
}

/**
 * 人物实体页全文。
 *
 * frontmatter 必须是 `type: entity` + `tags` 含 `character`：这是
 * `isCharacterEntityContent`（character-aura.ts:757-763）认出「可绑定人物页」
 * 的唯一判据，写错这一行绑定就过不了校验。
 */
export function buildProfilePageContent(
  name: string,
  body: string,
  date: Date = new Date(),
): string {
  const day = date.toISOString().slice(0, 10)
  return [
    "---",
    "type: entity",
    `title: ${titleScalar(name)}`,
    "tags: [character]",
    `created: ${day}`,
    `updated: ${day}`,
    "---",
    "",
    body.trim(),
    "",
  ].join("\n")
}

/** 绑定结果：成功 / 缺小传 / 其它错误。 */
export type ScenarioSoulBindResult =
  | { ok: true }
  | { ok: false; reason: "missing-profile" }
  | { ok: false; reason: "error"; message: string }

/** 补小传再绑的结果：无论成败都带上「这次建了没有」。 */
export type ScenarioSoulBindOutcome =
  | { ok: true; created: boolean }
  | { ok: false; created: boolean; message: string }

/** 这个错误是不是「缺人物小传」。 */
export function isMissingProfileError(error: unknown): boolean {
  return (
    error instanceof Error &&
    error.message.includes(CHARACTER_AURA_BINDING_BLOCK_MESSAGE)
  )
}

function toMessage(error: unknown): string {
  return error instanceof Error ? error.message : String(error)
}

/** 照常绑一次，并把失败原因分成三类交给界面。 */
export async function bindScenarioSoul(
  projectPath: string,
  characterName: string,
  auraId: string,
): Promise<ScenarioSoulBindResult> {
  try {
    await bindCharacterAura(projectPath, { characterName, auraId })
    return { ok: true }
  } catch (error) {
    if (isMissingProfileError(error)) return { ok: false, reason: "missing-profile" }
    return { ok: false, reason: "error", message: toMessage(error) }
  }
}

/**
 * 补一份最简小传，然后绑定。
 *
 * **已存在的人物页绝不覆盖** —— 用户手写的内容比我们拼的草稿重要得多，
 * 所以 fileExists 为真时直接跳过写入进入绑定。
 */
export async function createProfilePageThenBind(
  projectPath: string,
  characterName: string,
  auraId: string,
  body: string,
): Promise<ScenarioSoulBindOutcome> {
  const path = scenarioProfilePagePath(projectPath, characterName)
  let created = false

  try {
    const exists = await fileExists(path)
    if (!exists) {
      await createDirectory(`${normalizePath(projectPath)}/wiki/entities`)
      await writeFileAtomic(path, buildProfilePageContent(characterName, body))
      created = true
    }
  } catch (error) {
    return {
      ok: false,
      created: false,
      message: `写入人物页失败：${toMessage(error)}`,
    }
  }

  const result = await bindScenarioSoul(projectPath, characterName, auraId)
  if (result.ok) return { ok: true, created }
  if (result.reason === "missing-profile") {
    return {
      ok: false,
      created,
      message:
        "人物页已写入，但绑定仍被拒绝。请确认这份小传里写到了角色姓名，或先在「人物小传」里补全档案。",
    }
  }
  return { ok: false, created, message: result.message }
}

import { useTranslation } from "react-i18next"
import { Search, UserCog } from "lucide-react"

import { cn } from "@/lib/utils"
import type { ScenarioCandidate } from "@/lib/novel/story-simulation/scenario-types"

export interface ScenarioCharacterGridProps {
  candidates: ScenarioCandidate[]
  /** 已上台的角色 id；数组顺序即上台顺序，卡片上的序号取自这里。 */
  selectedIds: string[]
  /** 本次要选几个；选满后未上台的卡片会被淡化。 */
  characterCount: number
  /** 主持人 id；SCENARIO_MODERATOR_USER_ID 表示由用户本人担任。 */
  moderatorId: string

  searchThreshold?: number
  /** 点卡片：未上台则上台，已上台则下台。满员与否由调用方决定并给出提示。 */
  onToggle: (id: string) => void
  onSetModerator: (id: string) => void
  onBindSoul: (candidate: ScenarioCandidate) => void
}

const DEFAULT_SEARCH_THRESHOLD = 12

/**
 * 「舞台情景」候选角色网格（哑组件）
 *
 * 只负责呈现与上报：人数上限、上台顺序、主持人归属全部由父组件持有。
 * 拆出来是因为设置面板加上网格与弹窗入口后会超过 1100 行，一次读不完。
 */
export function ScenarioCharacterGrid({
  candidates,
  selectedIds,
  characterCount,
  moderatorId,
  searchThreshold = DEFAULT_SEARCH_THRESHOLD,
  onToggle,
  onSetModerator,
  onBindSoul,
}: ScenarioCharacterGridProps) {
  const { t } = useTranslation()
  const [query, setQuery] = useState("")

  const keyword = query.trim().toLowerCase()
  const visible = useMemo(() => {
    if (!keyword) return candidates
    return candidates.filter((c) =>
      `${c.name} ${c.auraName ?? ""}`.toLowerCase().includes(keyword),
    )
  }, [candidates, keyword])

  if (candidates.length === 0) return null
  const full = selectedIds.length >= characterCount

  return (
    <div className="flex flex-col gap-3">
      {candidates.length > searchThreshold && (
        <label className="flex items-center gap-2 rounded-lg border border-input px-2.5">
          <Search className="h-4 w-4 shrink-0 opacity-60" />
          <input
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            placeholder={t("storySimulation.scenarioCharacterSearch")}
            aria-label={t("storySimulation.scenarioCharacterSearch")}
            className="h-8 w-full bg-transparent text-sm outline-none"
          />
        </label>
      )}

      {visible.length === 0 ? (
        <p className="rounded-lg border border-dashed p-3 text-xs text-muted-foreground">
          {t("storySimulation.scenarioCharacterNoMatch", { query: query.trim() })}
        </p>
      ) : (
        <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
          {visible.map((c) => {
            const order = selectedIds.indexOf(c.id)
            const onStage = order >= 0
            const isMod = onStage && moderatorId === c.id
            return (
              <div
                key={c.id}
                className={cn(
                  "flex flex-col gap-2 rounded-lg border p-3 transition-colors",
                  onStage
                    ? "border-primary bg-primary/5 ring-2 ring-primary/20"
                    : "border-border hover:border-primary/50 hover:bg-accent/30",
                  // 已满且没上台：淡一点，让人一眼看出「再点也不会加进去」
                  !onStage && full && "opacity-60",
                )}
              >
                <div className="flex items-start justify-between gap-2">
                  <button
                    type="button"
                    onClick={() => onToggle(c.id)}
                    aria-pressed={onStage}
                    aria-label={t("storySimulation.scenarioToggleCharacter", {
                      name: c.name,
                    })}
                    className="flex min-w-0 flex-1 items-center gap-2 text-left"
                  >
                    {onStage ? (
                      <span className="grid h-5 w-5 shrink-0 place-items-center rounded-full bg-primary text-[0.625rem] font-semibold text-primary-foreground">
                        {order + 1}
                      </span>
                    ) : (
                      <span className="h-5 w-5 shrink-0 rounded-full border border-input" />
                    )}
                    <span className="truncate text-sm font-medium">{c.name}</span>
                    {isMod && (
                      <span className="shrink-0 rounded-full bg-primary/10 px-2 py-0.5 text-[0.625rem] font-medium text-primary">
                        {t("storySimulation.scenarioModerator")}
                      </span>
                    )}
                  </button>

                    <button
                      type="button"
                      onClick={() => onSetModerator(c.id)}
                      aria-label={t("storySimulation.scenarioSetModerator")}
                      title={t("storySimulation.scenarioSetModerator")}
                      className="shrink-0 rounded-md p-1 text-muted-foreground hover:bg-accent hover:text-foreground"
                    >
                      <UserCog className="h-4 w-4" />
                    </button>
                  )}
                </div>

                <div className="flex flex-wrap items-center gap-1.5 text-xs">
                  <button
                    type="button"
                    onClick={() => onBindSoul(c)}
                    title={
                      c.auraName
                        ? t("storySimulation.scenarioChangeSoul")
                        : t("storySimulation.scenarioBindSoul")
                    }
                    className={cn(
                      "rounded-full px-2 py-0.5",
                      c.auraName

                    )}
                  >

                  </button>

                    <span
                      key={p}
                      className="rounded-full bg-muted px-2 py-0.5 text-muted-foreground"
                    >
                      {p}
                    </span>
                  ))}
                </div>
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}

 * 「舞台情景」绑定灵魂弹窗
 *
 * ── 一个弹窗、两个步骤 ──
 * 步骤 1 选灵魂；步骤 2 只在 `bindCharacterAura` 明确报「缺人物小传」时出现。

 * 对本来能绑成功的角色弹出多余的补小传步骤。
 *

 * 那个弹窗是「本条消息的临时绑定」语义，文案写着「只对这一条消息生效」，
 * 而且是多选；本处要做的是**项目级持久绑定**，两者同形不同义。

export interface ScenarioSoulBindDialogProps {
  open: boolean
  projectPath: string
  /** 要绑定的角色；为 null 时弹窗内容为空。 */
  candidate: ScenarioCandidate | null
  onOpenChange: (open: boolean) => void
  /** 绑定或解绑成功后通知调用方刷新候选。 */
  onBound: () => void
}

export function ScenarioSoulBindDialog({
  open,
  projectPath,
  candidate,
  onOpenChange,
  onBound,
}: ScenarioSoulBindDialogProps) {
  const { t } = useTranslation()
  const [step, setStep] = useState<"pick" | "profile">("pick")
  const [auras, setAuras] = useState<CharacterAura[] | null>(null)
  const [query, setQuery] = useState("")
  const [pickedId, setPickedId] = useState("")
  const [draft, setDraft] = useState("")
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState<string | null>(null)

  // 每次打开都重读名单：用户可能刚在灵魂页建了一个灵魂
  useEffect(() => {
    if (!open) return
    setStep("pick")
    setQuery("")
    setPickedId(candidate?.auraId ?? "")
    setDraft("")
    setBusy(false)
    setError(null)
    setAuras(null)
    let current = true
    void listCharacterAuras(projectPath)
      .then((list) => {
        if (current) setAuras(list)
      })
      // 读不到就当空名单：空态会告诉用户去哪儿补，绝不能因此挡住弹窗
      .catch(() => {
        if (current) setAuras([])
      })
    return () => {
      current = false
    }
  }, [open, projectPath, candidate])

  const keyword = query.trim().toLowerCase()
  const visible = useMemo(() => {
    const list = auras ?? []
    if (!keyword) return list
    return list.filter((a) => a.name.toLowerCase().includes(keyword))
  }, [auras, keyword])

  if (!candidate) return null

  const handleBind = async () => {
    if (!pickedId || busy) return
    setBusy(true)
    setError(null)
    const result = await bindScenarioSoul(projectPath, candidate.name, pickedId)
    setBusy(false)
    if (result.ok) {
      onBound()
      onOpenChange(false)
      return
    }
    if (result.reason === "missing-profile") {
      // 只有这一条路径才切到补小传：草稿不调模型、不消耗 token
      setDraft(buildProfileDraftFromCandidate(candidate))
      setStep("profile")
      return
    }
    setError(result.message)
  }

  const handleCreateAndBind = async () => {
    if (busy || !draft.trim()) return
    setBusy(true)
    setError(null)
    const result = await createProfilePageThenBind(
      projectPath,
      candidate.name,
      pickedId,
      draft,
    )
    setBusy(false)
    if (result.ok) {
      onBound()
      onOpenChange(false)
      return
    }
    setError(result.message)
  }

  const handleUnbind = async () => {
    if (busy) return
    setBusy(true)
    setError(null)
    try {
      await unbindCharacterAura(projectPath, candidate.name)
      setBusy(false)
      onBound()
      onOpenChange(false)
    } catch (err) {
      setBusy(false)
      setError(err instanceof Error ? err.message : String(err))
    }
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent
        data-testid="scenario-soul-bind-dialog"
        className="flex max-h-[85vh] flex-col gap-3 overflow-y-auto sm:max-w-[720px]"
      >
        <DialogHeader className="shrink-0">
          <DialogTitle>
            {step === "pick"
              ? t("storySimulation.scenarioSoulDialogTitle", { name: candidate.name })
              : t("storySimulation.scenarioSoulProfileTitle")}
          </DialogTitle>
        </DialogHeader>

        {step === "pick" ? (
          <>
            <p className="shrink-0 text-sm text-muted-foreground">
              {t("storySimulation.scenarioSoulDialogDesc")}
            </p>

            <label className="flex shrink-0 items-center gap-2 rounded-md border px-2">
              <Search className="size-4 shrink-0 opacity-60" />
              <input
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                placeholder={t("storySimulation.scenarioSoulSearch")}
                aria-label={t("storySimulation.scenarioSoulSearch")}
                className="h-9 w-full bg-transparent text-sm outline-none"
              />
            </label>

            <div className="min-h-0 flex-1 overflow-y-auto" data-testid="scenario-soul-list">
              {auras === null ? (
                <p className="text-sm text-muted-foreground">
                  {t("storySimulation.scenarioSoulLoading")}
                </p>
              ) : visible.length === 0 ? (
                <p role="alert" className="text-sm">
                  {t("storySimulation.scenarioSoulEmpty")}
                </p>
              ) : (
                <div className="flex flex-col gap-1">
                  {visible.map((aura) => (
                    <label key={aura.id} className="flex items-center gap-2 py-1.5 text-sm">
                      <input
                        type="radio"
                        name="scenario-soul"
                        checked={pickedId === aura.id}
                        onChange={() => setPickedId(aura.id)}
                        aria-label={aura.name}
                      />
                      <span className="truncate" title={aura.name}>
                        {aura.name}
                      </span>
                    </label>
                  ))}
                </div>
              )}
            </div>

            {error && (
              <p role="alert" className="shrink-0 text-sm text-red-600 dark:text-red-400">
                {error}
              </p>
            )}

            <DialogFooter className="shrink-0">
              {candidate.auraId && (
                <Button variant="ghost" disabled={busy} onClick={() => void handleUnbind()}>
                  {t("storySimulation.scenarioSoulUnbind")}
                </Button>
              )}
              <Button variant="outline" onClick={() => onOpenChange(false)}>
                {t("storySimulation.back")}
              </Button>
              <Button disabled={!pickedId || busy} onClick={() => void handleBind()}>
                {busy
                  ? t("storySimulation.scenarioSoulBinding")
                  : t("storySimulation.scenarioSoulConfirm")}
              </Button>
            </DialogFooter>
          </>
        ) : (
          <>
            <p className="shrink-0 text-sm text-muted-foreground">
              {t("storySimulation.scenarioSoulProfileDesc", { name: candidate.name })}
            </p>

            <Textarea
              value={draft}
              onChange={(e) => setDraft(e.target.value)}
              rows={12}
              aria-label={t("storySimulation.scenarioSoulProfileTitle")}
              className="min-h-0 flex-1 font-mono text-xs"
            />

            {error && (
              <p role="alert" className="shrink-0 text-sm text-red-600 dark:text-red-400">
                {error}
              </p>
            )}

            <DialogFooter className="shrink-0">
              <Button
                variant="ghost"
                disabled={busy}
                onClick={() => {
                  setStep("pick")
                  setError(null)
                }}
              >
                {t("storySimulation.scenarioSoulProfileBack")}
              </Button>
              <Button
                disabled={busy || !draft.trim()}
                onClick={() => void handleCreateAndBind()}
              >
                {busy
                  ? t("storySimulation.scenarioSoulBinding")
                  : t("storySimulation.scenarioSoulProfileCreate")}
              </Button>
            </DialogFooter>
          </>
        )}
      </DialogContent>
    </Dialog>
  )
}
