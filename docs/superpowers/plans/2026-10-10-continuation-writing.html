章节「续写」实施计划

Goal

在章节正文编辑器里加「续写」：右键菜单首项与选中浮出工具栏首项都能触发，用户在浮层输入框里填方向（可留空），模型读当前章节正文续写 300–400 字，先流式显示在浮层里，满意再点「插入正文」写进正文。

import type { ChatMessage } from "@/lib/llm-providers"

/** 续写目标字数：提示词里的软目标，不做硬截断（用户原话「300 到 400 字左右」）。 */
export const CONTINUATION_TARGET_MIN_CHARS = 300
export const CONTINUATION_TARGET_MAX_CHARS = 400

/**
 * 送模正文上限。
 *
 * 超过就只送「开头 + 结尾」：用户要的是「能读到当前章节的内容，不需要读取
 * 全部内容」——整章 8000 字以内全送，超过就掐掉中段。掐中间而不是掐尾巴，
 * 是因为续写要接的是**结尾**的语气与动作，尾部绝不能丢。
 */
export const CONTINUATION_SOURCE_LIMIT_CHARS = 8000
export const CONTINUATION_SOURCE_HEAD_CHARS = 2000
export const CONTINUATION_SOURCE_TAIL_CHARS = 6000

export const CONTINUATION_TRUNCATION_MARK = "\n……（正文过长，此处省略）……\n"

/**
 * 输出上限，刻意宽松（约为目标字数的 5 倍）。
 *
 * 带思考的模型把思考 token 也算进这个额度，卡到 400 字对应的量级会让输出
 * 整段被截掉、用户拿到空结果。真正的字数控制交给提示词，这里只防跑飞。
 */
export const CONTINUATION_MAX_TOKENS = 2000
/** 续写要的是「接着写下去」而不是「最可能的下一句」，略高于默认值。 */
export const CONTINUATION_TEMPERATURE = 0.8

/** 输入框留空时，明确告诉模型「自行判断」，而不是丢一段空白让它猜。 */
export const CONTINUATION_EMPTY_INSTRUCTION = "未指定方向，请自行判断合理的推进"

/** 续写的插入位置。 */
export interface ContinuationAnchor {
  /** 正文偏移（**不含** frontmatter，也不含 `# 标题` 行）。 */
  at: number
  /** 右击 / 选中那一刻的正文快照，写入前用来检测正文是否已变化。 */
  bodySnapshot: string
}

/**
 * 要求编辑器把光标移到某个偏移。
 *
 * 不能复用 `highlightRequest`：那个是按文本 `indexOf` 找位置的
 * （wiki-editor.tsx:277），正文里出现重复句子时会定位到错的地方。
 */
export interface PendingEditorCaret {
  /** 目标光标偏移（正文偏移）。 */
  at: number
  /**
   * 变更计数。
   *
   * 光看 `at` 不够：连续两次插入可能算出同一个偏移，effect 的依赖数组判定
   * 「没变」就不重跑，光标会停在原地。带上 nonce 才能强制生效。
   */
  nonce: number
}

const CONTINUATION_SYSTEM_PROMPT = [
  "你是中文小说写作助手。任务：接着给定正文继续往下写。",
  "",
  `- 输出 ${CONTINUATION_TARGET_MIN_CHARS}–${CONTINUATION_TARGET_MAX_CHARS} 字。`,
  "- 只输出续写的正文。不要解释、不要复述已有内容、不要「好的」这类开场。",
  "- 承接最后一段的动作与情绪，自然衔接。",
  "- 保持人称、时态、叙事视角、文风、人物语气与给定正文一致。",
  "- 段落之间用一个换行分隔。不要自己加缩进，排版由应用统一处理。",
  "- 不要在结尾强行升华、总结或预告。",
].join("\n")

export function buildContinuationSource(body: string): string {
  if (body.length <= CONTINUATION_SOURCE_LIMIT_CHARS) return body
  return (
    body.slice(0, CONTINUATION_SOURCE_HEAD_CHARS)
    + CONTINUATION_TRUNCATION_MARK
    + body.slice(body.length - CONTINUATION_SOURCE_TAIL_CHARS)
  )
}

export function buildContinuationMessages(body: string, instruction: string): ChatMessage[] {
  const direction = instruction.trim() || CONTINUATION_EMPTY_INSTRUCTION
  const source = buildContinuationSource(body)
  const context = source.trim()
    ? `【本章已有正文】\n${source}`
    : "【本章已有正文】\n（本章正文为空，请从章首开始写。）"
  return [
    { role: "system", content: CONTINUATION_SYSTEM_PROMPT },
    { role: "user", content: `${context}\n\n【续写要求】\n${direction}` },
  ]
}

import { formatChapterWriting } from "@/lib/chapter-formatting"
export type ContinuationInsertResult =
  | { ok: true; body: string; caret: number }
  | { ok: false; reason: "changed" | "empty" }

/**
 * 把续写片段插到正文的锚点上。
 *
 * 与润色 / 去AI味 的 `replaceChapterBodySelection` 是**两套语义**：那边替换、
 * 这边插入。刻意不合并 —— 合并后浮层要同时承担「输入框」与「原文对比」两种
 * 形态，谁也说不清某个字段属于哪一边。
 *
 * 插入规则只做一件事：让续写自成一段。刻意**不**做「把插入点所在的段落切开、
 * 挪到续写后面再拼回来」这类聪明事 —— 那在用户眼里就是「我的字被动过了」。
 */
export function insertContinuationBody(
  currentBody: string,
  anchor: ContinuationAnchor,
  continuation: string,
): ContinuationInsertResult {
  /*
   * 先排版、再判空 —— 顺序和写法都不能改。
   *
   * 段首缩进是两个全角空格（U+3000），而 String.prototype.trim() 会把 U+3000
   * 一起去掉（它属于 Unicode 空白）。所以这里**只能拿 trim() 的结果判空**，
   * 绝不能拿它去拼正文，否则缩进会被静默吃掉。
   */
  const snippet = formatChapterWriting(continuation)
  if (!snippet.trim()) return { ok: false, reason: "empty" }
  if (anchor.bodySnapshot !== currentBody) return { ok: false, reason: "changed" }

  const at = Math.max(0, Math.min(anchor.at, currentBody.length))
  const before = currentBody.slice(0, at)
  const after = currentBody.slice(at)
  const lead = before && !before.endsWith("\n") ? "\n" : ""
  const tail = after && !after.startsWith("\n") ? "\n" : ""
  return {
    ok: true,
    body: `${before}${lead}${snippet}${tail}${after}`,
    caret: before.length + lead.length + snippet.length,
  }
}
为什么必须 formatChapterWriting 而不是自己加缩进

它同时做三件事：去掉行尾空白、补 　　 缩进、跑 pangu 中英间距，并保留本仓库的连字符 / 间隔号守卫（实测 苏-35飞过。 → 　　苏-35 飞过。，连字符没被 pangu 改成 苏 - 35）。自己拼字符串会漏掉后两件。

/**
 * 把浮层夹进视口。
 *
 * 做成纯函数（而不是在组件里量 DOM）有两个好处：能被直接单测；不必依赖
 * jsdom 里一堆返回 0 的 getBoundingClientRect。
 *
 * 优先放锚点**下方**：续写浮层是从「刚写完的那句话」触发的，压在正文上会
 * 挡住用户正要看的地方。下方放不下才翻到上方。
 */
export function clampContinuationPopoverPosition(
  anchor: { x: number; y: number },
  viewport: { width: number; height: number },
  size: { width: number; height: number } = {
    width: CONTINUATION_POPOVER_WIDTH,
    height: CONTINUATION_POPOVER_MAX_HEIGHT,
  },
  margin = 8,
): ContinuationPopoverPosition {
  const maxLeft = Math.max(margin, viewport.width - size.width - margin)
  const left = Math.min(Math.max(anchor.x - size.width / 2, margin), maxLeft)

  const below = anchor.y + CONTINUATION_POPOVER_GAP
  const above = anchor.y - size.height - CONTINUATION_POPOVER_GAP
  const top = below + size.height + margin <= viewport.height
    ? below
    : Math.max(margin, above)

  return { left, top }
}

import {
  CONTINUATION_POPOVER_MAX_HEIGHT,
  CONTINUATION_POPOVER_WIDTH,
  countContinuationChars,
  type ContinuationPopoverPosition,
} from "@/lib/novel/continuation"

export type ContinuationPhase = "idle" | "streaming" | "done" | "error"

export interface ContinuationPopoverProps {
  phase: ContinuationPhase
  position: ContinuationPopoverPosition
  instruction: string
  candidate: string
  errorMessage: string
  /** 模型是被输出上限截断的（finish reason 命中 length / max_tokens）。 */
  truncated: boolean
  onInstructionChange: (value: string) => void
  onStart: () => void
  onInsert: () => void
  onRegenerate: () => void
  onStop: () => void
  onClose: () => void

const BUTTON_CLASS =
  "rounded px-2 py-1 text-xs text-foreground transition-colors hover:bg-accent"

/**
 * 续写浮层。
 *
 * 刻意做成哑组件：不读 store、不发请求，只把状态画出来并把点击转成回调。
 * 这样它能在 jsdom 里被完整断言，而所有业务判断都留在 preview-panel 里。
 */
export function ContinuationPopover({
  phase,
  position,
  instruction,
  candidate,
  errorMessage,
  truncated,
  onInstructionChange,
  onStart,
  onInsert,
  onRegenerate,
  onStop,
  onClose,
}: ContinuationPopoverProps) {
  const streaming = phase === "streaming"
  const charCount = countContinuationChars(candidate)

  return (
    <div
      data-continuation-popover="true"
      role="dialog"
      aria-label="续写"
      className="fixed z-50 flex flex-col gap-2 rounded-md border bg-popover p-2 text-sm text-popover-foreground shadow-lg"
      style={{
        left: position.left,
        top: position.top,
        width: CONTINUATION_POPOVER_WIDTH,
        maxHeight: CONTINUATION_POPOVER_MAX_HEIGHT,
      }}
    >
      <div className="flex items-center justify-between gap-2 px-1">
        <div className="text-sm font-medium">续写</div>
        <button
          type="button"
          className="rounded p-1 text-muted-foreground hover:bg-accent hover:text-foreground"
          onClick={onClose}
          aria-label="关闭续写"
          title="关闭续写"
        >
          <X className="h-3.5 w-3.5" />
        </button>
      </div>

      <input
        type="text"
        data-continuation-instruction="true"
        value={instruction}
        disabled={streaming}
        placeholder="想写什么？可留空直接续写"
        onChange={(event) => onInstructionChange(event.target.value)}
        onKeyDown={(event) => {
          // 回车即续写，跟搜索框一致；但生成中不能再触发第二次。
          if (event.key === "Enter" && !streaming) onStart()
        }}
        className="w-full rounded border bg-background px-2 py-1 text-xs outline-none disabled:opacity-60"
      />

      {streaming ? (
        <div
          data-continuation-streaming="true"
          className="max-h-40 overflow-auto whitespace-pre-wrap rounded border bg-background px-2 py-1.5 text-xs leading-relaxed"
        >
          {candidate || <span className="text-muted-foreground">正在续写…</span>}
        </div>
      ) : null}

      {phase === "done" ? (
        <div
          data-continuation-candidate="true"
          className="max-h-40 overflow-auto whitespace-pre-wrap rounded border bg-background px-2 py-1.5 text-xs leading-relaxed"
        >
          {candidate}
        </div>
      ) : null}

      {phase === "error" ? (
        <div
          data-continuation-error="true"
          className="rounded border border-destructive/40 bg-destructive/10 px-2 py-1.5 text-xs text-destructive"
        >
          {errorMessage}
        </div>
      ) : null}

      {truncated && phase === "done" ? (
        <div className="px-1 text-xs text-amber-600 dark:text-amber-500">
          输出达到长度上限，已截断。可以插入，也可以重新生成。
        </div>
      ) : null}

      <div className="flex items-center justify-between gap-1 px-1">
        <span className="text-xs text-muted-foreground">{candidate ? `${charCount} 字` : ""}</span>
        <div className="flex items-center gap-1">
          {streaming ? (
            <button type="button" className={BUTTON_CLASS} onClick={onStop}>停止</button>
          ) : phase === "done" ? (
            <>
              <button type="button" className={BUTTON_CLASS} onClick={onInsert}>插入正文</button>
              <button type="button" className={BUTTON_CLASS} onClick={onRegenerate}>重新生成</button>
            </>
          ) : (
            <button type="button" className={BUTTON_CLASS} onClick={onStart}>续写</button>
          )}
        </div>
      </div>
    </div>
  )
}

function sliceBetween(source: string, startMarker: string, endMarker: string): string {
  const start = source.indexOf(startMarker)
  if (start < 0) throw new Error(`源码里找不到起点标记：${startMarker}`)
  const end = source.indexOf(endMarker, start + startMarker.length)
  if (end < 0) throw new Error(`源码里找不到终点标记：${endMarker}`)
  return source.slice(start, end)
}

/** 取一个「以 startMarker 开头、到它的依赖数组那一行为止」的代码块。 */
function callbackBlock(source: string, startMarker: string): string {
  const start = source.indexOf(startMarker)
  if (start < 0) throw new Error(`源码里找不到：${startMarker}`)
  /*
   * 依赖数组那一行长这样：`  }, [...]`（preview-panel 里缩进 2）
   * 或 `    }, [...]`（wiki-editor 的 effect 里缩进 4）。所以允许 2–6 个空格，
   * 只写死 2 个会匹配不到后者，然后一路找到别的函数的结尾 —— 断言看着通过，
   * 其实查的是另一段代码。
   *
   * 缩进必须写成 `[ \t]` 而不是 `\s`：`\s` 连换行一起匹配，配合 `m` 标志会
   * 跨行凑出「行首 + 若干空白 + }, [」，于是可能提前匹配到别的函数的结尾。
   */
  const end = /^[ \t]{2,6}\}, \[[^\n]*/m.exec(source.slice(start))
  if (!end || end.index === undefined) throw new Error(`找不到 ${startMarker} 的依赖数组结尾`)
  /*
   * 抽出来的块必须真的以 startMarker 开头、以 `}, [` 结尾。
   *
   * 终点行本身也要切进来（末尾那个 `+ end[0].length`）。少了它有两个后果，
   * 而且都是**静默**失效，不会报错：
   * ① 下面那句 `/\}, \[/.test(block)` 永远为假 —— 块切到终点行之前，所以
   *    连正确抽取都会抛「抽取结果不合理」；
   * ② 「依赖数组里不能有 value」那类 not.toContain 断言恒为真 —— 依赖数组
   *    根本不在块里，改坏了也测不出来。
   */
  const block = source.slice(start, start + end.index + end[0].length)
  if (!block.startsWith(startMarker) || !/\}, \[/.test(block)) {
    throw new Error(`${startMarker} 的代码块抽取结果不合理`)
  }
  return block
}
